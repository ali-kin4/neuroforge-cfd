\documentclass[preprint,12pt,a4paper]{elsarticle}

% elsarticle loads natbib itself -- do NOT load it again.
\input{preamble}


% tmlr.sty declares \sloppy; elsarticle does not. The single-column 12pt measure is
% narrow for this math-dense text, and without \sloppy the default tolerance leaves
% more loose lines, not fewer (measured: 10 against 0). File paths also break after
% "/" and "_" (see the shared preamble), so this build is free of overfull AND
% underfull boxes at LaTeX's default \hbadness.
\sloppy
\emergencystretch=4em

% elsarticle's run-in \paragraph appends a full stop to the heading
%   \def\els@bparagraph#1{\elsparagraph*{#1.}}
% and every \paragraph title in body.tex already ends in one, so the Elsevier
% build rendered 54 headings as "Title.." while the shared TMLR build (whose
% \paragraph adds nothing) rendered them correctly. Drop the class's period and
% let the shared source supply it, so both builds agree.
\makeatletter
\def\els@aparagraph[#1]#2{\elsparagraph[#1]{#2}}
\def\els@bparagraph#1{\elsparagraph*{#1}}
\makeatother


% elsarticle's \appendix relabels tables and figures "A.n" but never resets their
% counters, so the one appendix table printed as "Table A.7"; the guide numbers them
% Table A.1, Fig. A.1. Reset both, and give hyperref anchors that cannot collide with
% the main text's table.1.
\makeatletter
\g@addto@macro\appendix{%
  \setcounter{table}{0}\setcounter{figure}{0}%
  \gdef\theHtable{app.\Alph{section}.\arabic{table}}%
  \gdef\theHfigure{app.\Alph{section}.\arabic{figure}}}
\makeatother

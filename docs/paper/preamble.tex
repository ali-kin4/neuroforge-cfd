% Typesetting tolerance. Long monospace file paths used to be unbreakable, which
% forced loose lines around them; an earlier revision hid those by raising
% \hbadness/\vbadness to 10000. That is gone: the paths now break after "/" (the
% body writes \slash{} inside \texttt) and after "\_" (below, at the same penalty
% \slash uses, with no hyphen inserted), so both builds are free of overfull AND underfull boxes
% at LaTeX's default thresholds. \emergencystretch keeps overfull at zero.
\emergencystretch=3em
% (The kernel's own definition, plus the break; \let-ing the old \_ would recurse,
% because \DeclareRobustCommand also replaces the internal command it points to.)
% Each break point also carries a little stretch, as url.sty's \Urlmuskip does:
% monospace text has none of its own, so without it the spaces around a long path
% absorb all of a line's stretch. The glue vanishes when the line breaks there.
\makeatletter
\def\nf@pathbreak{\penalty\exhyphenpenalty\hskip\z@\@plus.25em\relax}
\DeclareRobustCommand{\_}{%
  \ifmmode\nfss@text{\textunderscore}\else\textunderscore\nf@pathbreak\fi}
\DeclareRobustCommand{\slash}{/\nf@pathbreak}
\makeatother
